cmake_minimum_required(VERSION 3.22)
project(portable_collision_check)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(rclcpp REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(check_model src/check_model.cpp)
ament_target_dependencies(check_model moveit_core moveit_ros_planning rclcpp)
target_link_libraries(check_model yaml-cpp)
install(TARGETS check_model DESTINATION lib/${PROJECT_NAME})
ament_package()
